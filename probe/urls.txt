import csv
def load(path, ticker):
    rows=[]; started=False
    with open(path, newline='', encoding='utf-8', errors='replace') as fh:
        for line in fh:
            line=line.rstrip('\n').rstrip('\r')
            if not started:
                if line.lower().startswith('"fund ticker"') or line.lower().startswith('fund ticker'):
                    started=True
                continue
            cells=next(csv.reader([line]))
            if not cells or not cells[0].strip(): continue
            if cells[0].strip()!=ticker: continue
            rows.append(cells)
    return rows
for ticker in ('NOBL','IGHG','TQQQ'):
    rows=load('/tmp/a.csv', ticker)
    mv=[abs(float(r[9])) for r in rows if r[9].strip()]

        print('   ', r[2], r[4][:38], 'mv', r[9], 'w_all %.4f%%' % (v/sum(mv)*100), 'w_nonnet %.4f%%' % (v/sum(mv_no_net)*100))
    for r in rows:
